### 节流防抖

```JavaScript
防抖
const debounce = (fn, delay) => {
  let timer = null
  return (...args) => {
    clearTimeout(timer)
    timer = setTimeout(() => fn.apply(this, args), delay)
  }
}

节流时间戳版本
const throttle = (fn, interval) => {
  let lastTime = 0
  return (...args) => {
    const now = Date.now()
    if (now - lastTime >= interval) {
      fn.apply(this, args)
      lastTime = now
    }
  }
}

节流定时器版本
const throttleTimer = (fn, delay) => {
  let timer = null
  return (...args) => {
    if (!timer) {
      timer = setTimeout(() => {
        fn(...args)
        timer = null
      }, delay)
    }
  }
}
```

### 解析 URL 参数

```javascript
function getUrlParams(url) {
  const params = {}
  const regex = /[?&]([^=#&]+)=([^&#]*)/g
  let match

  while ((match = regex.exec(url))) {
    const key = decodeURIComponent(match[1])
    const value = decodeURIComponent(match[2])
    params[key] = value
  }

  return params
}

或者
const url = 'https://example.com?name=John&age=30'
const params = url
  .split('?')[1]
  ?.split('&')
  .reduce((acc, cur) => {
    const [k, v] = cur.split('=')
    acc[k] = decodeURIComponent(v)
    return acc
  }, {})
// {name: "John", age: "30"}

// 使用示例
const url = 'https://example.com/?name=John&age=30&city=New%20York'
console.log(getUrlParams(url))
```

### JSONP

```javascript
function jsonp(url, callbackKey = 'callback') {
  const callbackName = `jsonp_${Date.now()}`
  const script = document.createElement('script')

  window[callbackName] = (data) => {
    delete window[callbackName]
    document.body.removeChild(script)
    return data
  }

  script.src = `${url}${
    url.includes('?') ? '&' : '?'
  }${callbackKey}=${callbackName}`
  document.body.appendChild(script)
}
```

### Promise

```javascript
class MyPromise {
    construction(executor){
        executor(this.resolve, this.reject);
    }

    // 状态 pendding fulilled reject
    status = 'pendding'
     // 成功之后的值
    value = null;
    // 失败之后的原因
    reason = null;
    successStack = []
    rejectStack = []


    resolve = (value)=>{
        if(status === 'pedding'){
           this.status = 'fullfill'
           this.value = value
           while(this.successStack.length){
               this.successStack.shift()(value)
           }
        }
    }

    reject = (reason) =>{
        if(status === 'pedding'){
           this.status = 'faild'
           this.reason= reason
           while(this.rejectStack .length){
               this.rejectStack .shift()(reason)
           }
        }
    }

    then = (onFulFilled,onReject)=>{
        if(this.status === 'fulfilled'){
            onFulFilled(this.value)
        } else if(this.status === 'reject'){
            onReject(this.reason)
        } else {
            this.successStack.push(onFulFilled)
            this.rejectStack.push(onReject)
        }
    }
}
export MyPromies
```

### memoize

这个方法为一个 function 添加返回的缓存，如果传参相同，则优先取缓存的值。

比如 const memoizeFun = memoize(( value ) => console.log(value))

实际调用的还是里面的函数，只是为它实现一个缓存效果。

比如

```javascript
const obj= { a:1 },
memoizeFun(obj.a) 输出1
obj.a = 2
memoizeFun(obj.a) 还是输出1
```

那么来实现这个函数

```javascript
function memoize(func,resolver){
    const memoized= (...args)=>{
        // resolver 如果传了，则用 resolver(...args) 作为 key，官方文档要求，否则使用第一个参数
        const key = resolver ？ resolver(...args) : args[0]

        if (memoized.cache.has(key)) {
            // 直接返回结果
            return memoized.cache.get(key)
        } else {
            // 缓存结果
            const result = func.apply(this, args)
            memoized.cache.set(key , result )
            return result
        }
    }

    // 定义 WeakMap 缓存结果
    memoized.cache = new WeakMap()

    return memoize
}
```
